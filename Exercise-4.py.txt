class FollowerMatrix:
    def __init__(self, n):
        self.n = n
        self.matrix = [[False for _ in range(n)] for _ in range(n)]

    
    def follow(self, follower, followee):
        if follower != followee:
            self.matrix[follower][followee] = True

  
    def unfollow(self, follower, followee):
        self.matrix[follower][followee] = False

   
    def is_following(self, follower, followee):
        return self.matrix[follower][followee]

    
    def get_followers(self, user):
        return [i for i in range(self.n) if self.matrix[i][user]]

    
    def get_following(self, user):
        return [j for j in range(self.n) if self.matrix[user][j]]


    def mutual_follows(self):
        pairs = []
        for i in range(self.n):
            for j in range(i + 1, self.n):
                if self.matrix[i][j] and self.matrix[j][i]:
                    pairs.append((i, j))
        return pairs

  
    def influence_score(self, user):
        followers = len(self.get_followers(user))
        following = len(self.get_following(user))
        return (followers + following) / self.n
